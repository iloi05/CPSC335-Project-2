def min_boats(people, limit):

    sort(people)

    left = 0
    right = len(people) - 1
    boats = 0

    if people[right] > limit:
        return -1

    while left <= right:
        if people[left] + people[right] <= limit:
            left = left + 1

        right = right - 1
        boats = boats + 1

    return boats
