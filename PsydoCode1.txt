def shortest_path(grid, start, target):

    rows = len(grid)
    cols = len(grid[0])

    sx, sy = start
    tx, ty = target

    # if start or target is blocked
    if grid[sx][sy] == 1 or grid[tx][ty] == 1:
        return -1

    queue = [(sx, sy, 0)]      # (row, col, distance)
    visited = {(sx, sy)}

    directions = [(1,0), (-1,0), (0,1), (0,-1)]

    while queue is not empty:

        x, y, dist = queue.popleft()

        if (x, y) == (tx, ty):
            return dist

        for dx, dy in directions:

            nx = x + dx
            ny = y + dy

            if (0 <= nx < rows and
                0 <= ny < cols and
                grid[nx][ny] == 0 and
                (nx, ny) not in visited):

                add (nx, ny) to visited
                add (nx, ny, dist + 1) to queue

    return -1
